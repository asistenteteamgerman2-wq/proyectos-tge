  function bestCrop(boxes, W, H) {
    const pad = 0.025, bx = boxes.slice(0, 4).map(b => ({ x0: b.x0 - pad, y0: b.y0 - pad, x1: b.x1 + pad, y1: b.y1 + pad }));
    let best = null;
    const walk = (i, r) => {
      if (r.x1 - r.x0 <= 0 || r.y1 - r.y0 <= 0) return;
      if (i === bx.length) { const a = (r.x1 - r.x0) * (r.y1 - r.y0); if (!best || a > best.a) best = { ...r, a }; return; }
      const b = bx[i];
      if (b.x1 <= r.x0 || b.x0 >= r.x1 || b.y1 <= r.y0 || b.y0 >= r.y1) return walk(i + 1, r);
      walk(i + 1, { ...r, y1: Math.min(r.y1, b.y0) }); walk(i + 1, { ...r, y0: Math.max(r.y0, b.y1) });
      walk(i + 1, { ...r, x1: Math.min(r.x1, b.x0) }); walk(i + 1, { ...r, x0: Math.max(r.x0, b.x1) });
    };
    walk(0, { x0: 0, y0: 0, x1: 1, y1: 1 });
    if (!best) return null;
    let w = (best.x1 - best.x0) * W, hh = (best.y1 - best.y0) * H, cx = (best.x0 + best.x1) / 2 * W, cy = (best.y0 + best.y1) / 2 * H;
    if (w / hh > 2) w = hh * 2; if (hh / w > 1.4) hh = w * 1.4; // keep a usable photo shape
    if (w * hh < W * H * 0.5 || Math.min(w, hh) < 320) return null;
    return { x: Math.round(cx - w / 2), y: Math.round(cy - hh / 2), w: Math.round(w), h: Math.round(hh) };
  }

          o[i + k] = (hz + vt) / 2;
        }
        o[i + 3] = 255;

      for (const n of names) { const nn = pnorm(n); if (!nn) continue;
