function initLanding(root) {
  const lp = root.querySelector(".lp");
  if (!lp) return;
  lp.addEventListener("click", ev => {
    const b = ev.target.closest("[data-zoom]");
    const z = lp.querySelector(".zoom");
    if (z && ev.target.closest(".zoom")) {
      const nav = ev.target.closest("[data-nav]");
      if (nav) { z._go(Number(nav.dataset.nav)); return; }
      if (ev.target.tagName !== "IMG") z.remove();
      return;
    }
    if (!b) {
      const a = ev.target.closest('a[href^="#lp-"]');

    const key = e => { if (!d.isConnected) { removeEventListener("keydown", key); return; } if (e.key === "Escape") d.remove(); else if (e.key === "ArrowLeft") d._go(-1); else if (e.key === "ArrowRight") d._go(1); };

  async function imgToCanvas(img) {
    const c = document.createElement("canvas");
    if (img.bitmap) { c.width = img.bitmap.width; c.height = img.bitmap.height; c.getContext("2d").drawImage(img.bitmap, 0, 0); c._raw = true; return c; }
    const { width: w, height: hh, data, kind } = img;
    if (!w || !hh || !data) return null;
    c.width = w; c.height = hh;
    const ctx = c.getContext("2d"), id = ctx.createImageData(w, hh), px = id.data;
    if (kind === 3) px.set(data.subarray ? data.subarray(0, px.length) : data);
    else if (kind === 2) { for (let i = 0, j = 0; j < w * hh * 3; i += 4, j += 3) { px[i] = data[j]; px[i + 1] = data[j + 1]; px[i + 2] = data[j + 2]; px[i + 3] = 255; } }
    else return null;
    ctx.putImageData(id, 0, 0);
    return c;
  }
  // Mide si una imagen sirve como foto: descarta negras, blancas, planas o recortes con fondo transparente.
  function photoStats(c) {
    const S2 = 64, t = document.createElement("canvas"); t.width = S2; t.height = S2;
    const x = t.getContext("2d", { willReadFrequently: true }); x.drawImage(c, 0, 0, S2, S2);
    const d = x.getImageData(0, 0, S2, S2).data, n = S2 * S2;
    let sum = 0, sum2 = 0, transp = 0, rg = 0, yb = 0; const L = new Float32Array(n);
    for (let i = 0, k = 0; k < n; i += 4, k++) {
      if (d[i + 3] < 200) transp++;
      const l = 0.299 * d[i] + 0.587 * d[i + 1] + 0.114 * d[i + 2]; L[k] = l; sum += l; sum2 += l * l;
      rg += Math.abs(d[i] - d[i + 1]); yb += Math.abs((d[i] + d[i + 1]) / 2 - d[i + 2]);
    }
    const mean = sum / n, std = Math.sqrt(Math.max(0, sum2 / n - mean * mean));
    let flat = 0; for (let k = 0; k < n; k++) if (Math.abs(L[k] - mean) < 10) flat++;
    const color = (rg + yb) / n;
    const ok = transp / n < 0.25 && mean > 28 && mean < 238 && std > 16 && flat / n < 0.7;
    return { ok, score: std + color * 0.6 + (c.width / c.height >= 1.2 ? 10 : 0), mean, std };
  }
  function shrink(src, max) {
    const k = Math.min(1, max / Math.max(src.width, src.height));
    if (k === 1) return src;
    const c = document.createElement("canvas"); c.width = Math.round(src.width * k); c.height = Math.round(src.height * k);
    c.getContext("2d").drawImage(src, 0, 0, c.width, c.height); return c;
  }

    const doc = await lib.getDocument({ data: new Uint8Array(await file.arrayBuffer()) }).promise;
    let text = "";
    const n = Math.min(doc.numPages, 30);
    for (let i = 1; i <= n; i++) {
      const pg = await doc.getPage(i);
      const tc = await pg.getTextContent();
      text += `\n--- Página ${i} ---\n` + tc.items.map(it => it.str).join(" ");
      if (text.length > 60000) break;
    }
    const photos = [], seen = new Set();
    const OPS = lib.OPS;
    for (let i = 1; i <= Math.min(doc.numPages, 24) && photos.length < 16; i++) {
      setStatus(`Buscando fotos en el brochure (página ${i} de ${Math.min(doc.numPages, 24)})…`);
      const pg = await doc.getPage(i);
      let ops; try { ops = await pg.getOperatorList(); } catch { continue; }
      for (let k = 0; k < ops.fnArray.length && photos.length < 16; k++) {
        if (ops.fnArray[k] !== OPS.paintImageXObject) continue;
        const id = ops.argsArray[k][0];
        if (seen.has(id)) continue; seen.add(id);
        const store = String(id).startsWith("g_") ? pg.commonObjs : pg.objs;
        const img = await new Promise(r => { try { store.get(id, r); } catch { r(null); } setTimeout(() => r(null), 4000); });
        if (!img) continue;
        const w = img.width || img.bitmap?.width || 0, hh = img.height || img.bitmap?.height || 0;
        const ratio = w / Math.max(1, hh);
        if (Math.min(w, hh) < 350 || w * hh < 250000 || ratio > 3 || ratio < 0.33) continue;
        const c = await imgToCanvas(img); if (!c) continue;
        const st = photoStats(c); if (!st.ok) continue;
        const fl = document.createElement("canvas"); fl.width = c.width; fl.height = c.height;
        const fx = fl.getContext("2d"); fx.fillStyle = "#fff"; fx.fillRect(0, 0, fl.width, fl.height); fx.drawImage(c, 0, 0);
        const out = shrink(fl, 1800); out._score = st.score; photos.push(out);
      }
    }
    let pages = null;
    const renderPages = async () => {
      if (pages) return pages;
      pages = [];
      for (let i = 1; i <= Math.min(doc.numPages, 6); i++) {
        const pg = await doc.getPage(i), v1 = pg.getViewport({ scale: 1 });
        const vp = pg.getViewport({ scale: Math.min(2, 1400 / Math.max(v1.width, v1.height)) });
        const c = document.createElement("canvas"); c.width = Math.round(vp.width); c.height = Math.round(vp.height);
        await pg.render({ canvasContext: c.getContext("2d"), viewport: vp }).promise;
        pages.push(await toJpeg(c, 0.8));
      }
      return pages;
    };
    return { text: text.slice(0, 60000), photos, renderPages };
  }
