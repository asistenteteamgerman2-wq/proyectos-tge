  function apiSample() {
    const key = apiKey(); if (!key) return null;
    const MODELS = { quick: "claude-haiku-4-5-20251001", default: "claude-sonnet-5-5", complex: "claude-opus-5-5" };
    const b64 = b => new Promise(ok => { const fr = new FileReader(); fr.onload = () => ok(String(fr.result).split(",")[1]); fr.readAsDataURL(b); });
    async function call(input, opts = {}) {
      const content = [];
      for (const im of opts.images || []) content.push({ type: "image", source: { type: "base64", media_type: im.type || "image/jpeg", data: await b64(im) } });
      content.push({ type: "text", text: String(input) });
      let r;
      try {
        r = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "content-type": "application/json", "x-api-key": key, "anthropic-version": "2023-06-01", "anthropic-dangerous-direct-browser-access": "true" },
          body: JSON.stringify({ model: MODELS[opts.modelTier] || MODELS.default, max_tokens: 4096, messages: [{ role: "user", content }] }) });
      } catch (x) { throw { code: "network", message: "Sin conexión con Claude" }; }
      if (r.status === 401 || r.status === 403) throw { code: "not_granted", message: "La clave de API no es válida" };
      if (r.status === 429) throw { code: "rate_limited" };
      if (!r.ok) throw { code: "api_error", message: "Error " + r.status };
      const j = await r.json();
      return { text: (j.content || []).filter(c => c.type === "text").map(c => c.text).join(""), truncated: j.stop_reason === "max_tokens" };
    }
    const fn = (input, opts) => call(input, opts);
    fn.json = async (input, opts) => {
      const { text } = await call(input + "\n\nResponde únicamente con el JSON, sin texto antes ni después.", opts);
      const a = text.indexOf("{"), b = text.lastIndexOf("}");
      try { return JSON.parse(text.slice(a, b + 1)); } catch (x) { throw { code: "invalid_json" }; }
    };
    fn.limits = async () => ({ images: { maxCount: 8 } });
    return fn;
  }

      toast("Se cargaron " + SEED.proyectos.length + " proyectos");
    };
    db.collection("proyectos").onSnapshot(snap => {
      seedIfEmpty(snap);
      S.projects = snap.docs.filter(d => d.exists).map(d => {
        const v = d.data() || {};
        return { id: d.id, ...v,
          tipologias: Array.isArray(v.tipologias) ? v.tipologias.map(t => ({ hab: Number(t.hab) || 1, desde: Number(t.desde) || 0, banos: Number(t.banos) || 0, m2: Number(t.m2) || 0 })) : [],
          fotos: Array.isArray(v.fotos) ? v.fotos.filter(x => typeof x === "string") : [] };
      });
      S.loaded = true;
      render();
      if (location.hash.startsWith("#p-") && S.openId !== location.hash.slice(3)) route();
    }, () => { S.loaded = true; render(); });
  })();
})();
